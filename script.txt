// 3. A MÁGICA: O JS cria o HTML dinamicamente

cardapio.forEach(prato => {
  
  // Note que aqui dentro das crases (`) a gente escreve o HTML normal!
  // Usamos ${} para puxar os dados do Array.

// LÓGICA DO ACCORDION (CLIQUE PARA EXPANDIR)
// ==========================================

// O "container" é a sua <section> que já selecionamos lá em cima.
// Vamos escutar qualquer clique que aconteça dentro dela:
container.addEventListener('click', function(event) {
  
  // Confirma se o que foi clicado foi realmente o botão "+"
  if (event.target.classList.contains('btn-expandir')) {
    
    // Pega o card inteiro onde esse botão específico foi clicado
    const cardClicado = event.target.closest('.card-produto');
    
    // Descobre se esse card específico já está aberto
    const jaEstaAtivo = cardClicado.classList.contains('ativo');
    
    // PASSO 1: FECHAR TODOS
    // Varre a tela procurando todos os cards e remove a classe '.ativo' de todos eles.
    // Também garante que todos os botões voltem a ter o texto "+"
    document.querySelectorAll('.card-produto').forEach(card => {
      card.classList.remove('ativo');
      card.querySelector('.btn-expandir').textContent = '+';
    });
    
    // PASSO 2: ABRIR O CLICADO (se ele já não estava aberto antes)
    // Como fechamos todos no passo anterior, se o card clicado estava fechado, 
    // nós adicionamos a classe nele para abrir e mudamos o texto do botão para "-"
    if (!jaEstaAtivo) {
      cardClicado.classList.add('ativo');
      event.target.textContent = '-';
    }
  }
});
});
